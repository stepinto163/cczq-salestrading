2. c# 调用js网页 注意： 方法固定cef.subscribeCallback ，用于获得订阅回调 

<script type="text/javascript">
    // 等待对象注入
    function waitForBridge(callback) {
        const interval = setInterval(() => {
            if (typeof cef !== 'undefined') {
                clearInterval(interval);
                callback(cef);
            }
        }, 100);
    }
    waitForBridge((cef) => {
        console.log("cef 已注入");
        window.cefCallback= function (json) {
            const data = JSON.parse(json);
            console.log("收到回调:", data);
            //document.getElementById("price").innerText = data.price;
        };
         // 注册回调到 C# 桥接对象,WebFinq 第二个参数固定为 WebFQ
            cef.subscribeCallback("window.cefCallback","C# Js Call");
    });
 </script>
